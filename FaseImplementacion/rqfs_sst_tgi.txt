// RQF049: Calcular porcentaje de cumplimiento a capacitaciones por trabajador
db.asistencias.aggregate([
  { $match: { "trabajador._id": 1 } },
  { $group: {
      _id: "$trabajador._id",
      nombreTrabajador: { $first: "$trabajador.nombreTrabajador" },
      totalCitados:    { $sum: 1 },
      totalAsistieron: { $sum: { $cond: ["$asistio", 1, 0] } }
  }},
  { $project: {
      _id: 0,
      idTrabajador: "$_id",
      nombreTrabajador: 1,
      totalCitados: 1,
      totalAsistieron: 1,
      porcentajeCumplimiento: {
        $round: [
          { $multiply: [{ $divide: ["$totalAsistieron", "$totalCitados"] }, 100] },
          2
        ]
      }
  }}
])

// RQF057: Generar reporte completo por rango de fechas
db.programaciones.aggregate([
  { $match: { fechaProgramada: { $gte: "2026-03-01", $lte: "2026-04-30" } } },
  { $lookup: {
      from: "capacitaciones",
      localField: "idCapacitacion",
      foreignField: "_id",
      as: "capacitacion"
  }},
  { $unwind: "$capacitacion" },
  { $lookup: {
      from: "asistencias",
      localField: "_id",
      foreignField: "idProgramacion",
      as: "asistencias"
  }},
  { $project: {
      _id: 0,
      idProgramacion: "$_id",
      fechaProgramada: 1,
      horaProgramada: 1,
      lugarProgramado: 1,
      modalidadProgramada: 1,
      estadoProgramacion: 1,
      tema: "$capacitacion.tema",
      programa: "$capacitacion.programa.nombrePrograma",
      fasePHVA: "$capacitacion.programa.fasePHVA",
      responsables: "$capacitacion.responsables.nombreResponsable",
      accidentesAsociados: "$capacitacion.accidentesAsociados",
      aforoProgramado: "$asistenciaProgramada",
      totalCitados:    { $size: "$asistencias" },
      totalAsistieron: {
        $size: { $filter: { input: "$asistencias", as: "a", cond: "$$a.asistio" } }
      },
      porcentajeCumplimiento: {
        $cond: [
          { $eq: [{ $size: "$asistencias" }, 0] },
          0,
          { $round: [
              { $multiply: [
                  { $divide: [
                      { $size: { $filter: { input: "$asistencias", as: "a", cond: "$$a.asistio" } } },
                      { $size: "$asistencias" }
                  ]},
                  100
              ]},
              2
          ]}
        ]
      }
  }},
  { $sort: { fechaProgramada: 1 } }
])
