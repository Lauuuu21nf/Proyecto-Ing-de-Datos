// --- Porcentaje de cumplimiento de asistencia por rango de fecha en un programa ---
// Pipeline de agregacion con $lookup para cruzar asistencias con programaciones
db.asistencias.aggregate([
  { $match: { idPrograma: 7 } },
  { $lookup: {
    from: 'programaciones',
    localField: 'idProgramacion',
    foreignField: '_id',
    as: '_prgm'
  }},
  { $unwind: '$_prgm' },
  { $match: {
    '_prgm.fechaProgramada': { $gte: '2026-01-01', $lte: '2026-12-31' }
  }},
  { $group: {
    _id: '$idPrograma',
    numAsistentes: { $sum: { $cond: ['$asistio', 1, 0] } },
    asistenciaProgramada: { $sum: 1 }
  }},
  { $lookup: {
    from: 'programas',
    localField: '_id',
    foreignField: '_id',
    as: 'p'
  }},
  { $unwind: '$p' },
  { $project: {
    _id: 0,
    nombrePrograma: '$p.nombrePrograma',
    numAsistentes: 1,
    asistenciaProgramada: 1,
    porcentajeCumplimiento: {
      $cond: [
        { $eq: ['$asistenciaProgramada', 0] }, 0,
        { $round: [{ $multiply: [{ $divide: ['$numAsistentes', '$asistenciaProgramada'] }, 100] }, 2] }
      ]
    }
  }}
]);

// --- Estadisticas por trimestre para un año dado ---
// Se calculan: total programadas, total realizadas, porcentaje de cumplimiento y accidentes
var anio = 2026;
for (var trim = 1; trim <= 4; trim++) {
  var mesInicio = (trim - 1) * 3 + 1;
  var mesFin = trim * 3;
  var inicio = anio + '-' + (mesInicio < 10 ? '0' : '') + mesInicio + '-01';
  var ultimoDia = new Date(anio, mesFin, 0).getDate();
  var fin = anio + '-' + (mesFin < 10 ? '0' : '') + mesFin + '-' + (ultimoDia < 10 ? '0' : '') + ultimoDia;

  var progs = db.programaciones.find({
    fechaProgramada: { $gte: inicio, $lte: fin }
  }).toArray();

  var totalProgramadas = progs.length;
  var totalRealizadas = progs.filter(function(p) { return p.estadoProgramacion === 'realizada'; }).length;

  var idsCap = [];
  progs.forEach(function(p) {
    if (idsCap.indexOf(p.idCapacitacion) === -1) idsCap.push(p.idCapacitacion);
  });

  var totalAT = 0;
  if (idsCap.length > 0) {
    var caps = db.capacitaciones.find({ _id: { $in: idsCap } }).toArray();
    caps.forEach(function(c) { totalAT += (c.accidentesAsociados || 0); });
  }

  var pctCumplimiento = totalProgramadas > 0
    ? Math.round((totalRealizadas / totalProgramadas) * 10000) / 100
    : 0;

  print('--- Trimestre ' + trim + ' de ' + anio + ' ---');
  print('  Total programadas: ' + totalProgramadas);
  print('  Total realizadas: ' + totalRealizadas);
  print('  Porcentaje cumplimiento: ' + pctCumplimiento + '%');
  print('  Accidentes de trabajo asociados: ' + totalAT);
}
